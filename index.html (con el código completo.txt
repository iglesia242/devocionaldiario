  <script>
    const totalDays = 365;
    let currentDay = 1;
    let completedDays = new Set(JSON.parse(localStorage.getItem('completedDays_365') || '[]'));
    let isPlaying = false;
    let speechSyn = null;

    const otBooks = [
      "Génesis", "Éxodo", "Levítico", "Números", "Deuteronomio",
      "Josué", "Jueces", "Rut", "1 Samuel", "2 Samuel", "1 Reyes", "2 Reyes",
      "1 Crónicas", "2 Crónicas", "Esdras", "Nehemías", "Ester",
      "Job", "Salmos", "Proverbios", "Eclesiastés", "Cantares",
      "Isaías", "Jeremías", "Lamentaciones", "Ezequiel", "Daniel",
      "Oseas", "Joel", "Amós", "Abdías", "Jonás", "Miqueas",
      "Nahúm", "Habacuc", "Sofonías", "Hageo", "Zacarías", "Malaquías"
    ];

    const ntBooks = [
      "Mateo", "Marcos", "Lucas", "Juan", "Hechos",
      "Romanos", "1 Corintios", "2 Corintios", "Gálatas", "Efesios",
      "Filipenses", "Colosenses", "1 Tesalonicenses", "2 Tesalonicenses",
      "1 Timoteo", "2 Timoteo", "Tito", "Filemón", "Hebreos",
      "Santiago", "1 Pedro", "2 Pedro", "1 Juan", "2 Juan",
      "3 Juan", "Judas", "Apocalipsis"
    ];

    window.addEventListener('DOMContentLoaded', () => {
      renderDaysGrid();
      renderBibleBooks();
      loadDayContent(1);
      updateGlobalProgress();
    });

    function renderDaysGrid() {
      const container = document.getElementById('daysGrid');
      container.innerHTML = '';
      for (let i = 1; i <= totalDays; i++) {
        const isDone = completedDays.has(i);
        const isActive = i === currentDay;
        const btn = document.createElement('button');
        btn.onclick = () => loadDayContent(i);
        btn.className = `p-2 rounded-lg text-xs font-bold transition-all flex flex-col items-center justify-center aspect-square ${
          isActive ? 'bg-royal-800 text-white border-2 border-gold-400 shadow scale-105' :
          isDone ? 'bg-emerald-100 text-emerald-800 border border-emerald-300' :
          'bg-white text-slate-700 border border-slate-200 hover:bg-slate-100'
        }`;
        btn.innerHTML = `<span>${i}</span>${isDone ? '<i class="fa-solid fa-check text-[9px] text-emerald-600 mt-0.5"></i>' : ''}`;
        container.appendChild(btn);
      }
    }

    function renderBibleBooks() {
      const otContainer = document.getElementById('otBooksList');
      otContainer.innerHTML = otBooks.map(b => `<span class="px-2 py-1 bg-white border border-amber-200 rounded text-amber-900 font-semibold">${b}</span>`).join('');

      const ntContainer = document.getElementById('ntBooksList');
      ntContainer.innerHTML = ntBooks.map(b => `<span class="px-2 py-1 bg-white border border-amber-200 rounded text-amber-900 font-semibold">${b}</span>`).join('');
    }

    function toggleBibleLibrary() {
      const panel = document.getElementById('bibleLibraryPanel');
      const btnText = document.getElementById('bibleLibraryBtnText');
      panel.classList.toggle('hidden');
      btnText.innerText = panel.classList.contains('hidden') ? "Abrir Biblia Completa" : "Cerrar Biblia";
    }

    function loadDayContent(dayNum) {
      if (isPlaying) stopAudio();
      currentDay = parseInt(dayNum);

      document.getElementById('currentDayBadge').innerText = `DÍA ${currentDay} DE 365`;
      document.getElementById('currentDayTitle').innerText = `Estudio Bíblico y Devocional - Día ${currentDay}`;
      
      document.getElementById('otPassageTitle').innerText = `Antiguo Testamento (Día ${currentDay})`;
      document.getElementById('otPassageText').innerText = `Lectura correspondiente al día ${currentDay} en el Antiguo Testamento. Medita en la fidelidad de Dios y sus mandamientos perpetuos para tu vida y familia.`;

      document.getElementById('ntPassageTitle').innerText = `Nuevo Testamento (Día ${currentDay})`;
      document.getElementById('ntPassageText').innerText = `Lectura correspondiente al día ${currentDay} en el Nuevo Testamento. Contempla la gracia, el amor y la obra redentora de Jesucristo nuestro Señor.`;

      document.getElementById('devotionalTitle').innerText = `El Propósito Divino en el Día ${currentDay}`;
      document.getElementById('devotionalBody').innerText = `Cada jornada en la presencia del Señor fortalece nuestro espíritu. Hoy recordamos que el Creador sustenta nuestros pasos y nos llama a vivir en santidad y obediencia.`;
      document.getElementById('devotionalApp').innerText = `"Aplica la palabra de hoy perdonando a tu prójimo y caminando con integridad en tu labor diaria."`;

      document.getElementById('pastorTitle').innerText = `Reflexión Pastoral: Caminando en Victoria`;
      document.getElementById('pastorBody').innerText = `Amados hermanos de la Iglesia Cristiana Iré Por Ti y estudiantes del Instituto Teológico Nehemías, el Señor nos capacita hoy para ser luz en medio de cualquier dificultad. No teman, porque Su mano poderosa nos sostiene.`;

      document.getElementById('prayerText').innerText = `"Padre Celestial, te damos gracias por el don de la vida y por tu Palabra viva. Guía nuestros pasos en este día ${currentDay}, danos sabiduría, fortalece nuestra fe y permítenos reflejar tu amor a todos los que nos rodean. En el nombre de Jesús, Amén."`;

      updateCompletionBtnUI();
      renderDaysGrid();
    }

    function toggleCurrentDayCompletion() {
      if (completedDays.has(currentDay)) {
        completedDays.delete(currentDay);
      } else {
        completedDays.add(currentDay);
      }
      localStorage.setItem('completedDays_365', JSON.stringify([...completedDays]));
      updateCompletionBtnUI();
      renderDaysGrid();
      updateGlobalProgress();
    }

    function toggleCompleteAll() {
      if (completedDays.size === totalDays) {
        completedDays.clear();
      } else {
        for (let i = 1; i <= totalDays; i++) completedDays.add(i);
      }
      localStorage.setItem('completedDays_365', JSON.stringify([...completedDays]));
      updateCompletionBtnUI();
      renderDaysGrid();
      updateGlobalProgress();
    }

    function updateCompletionBtnUI() {
      const btn = document.getElementById('completeDayBtn');
      const text = document.getElementById('completeBtnText');
      if (completedDays.has(currentDay)) {
        btn.className = "px-5 py-2.5 rounded-xl font-bold text-xs flex items-center gap-2 shadow transition-all bg-slate-200 text-slate-700 hover:bg-slate-300";
        text.innerText = "Día Completado (Desmarcar)";
      } else {
        btn.className = "px-5 py-2.5 rounded-xl font-bold text-xs flex items-center gap-2 shadow transition-all bg-emerald-600 text-white hover:bg-emerald-700";
        text.innerText = "Marcar como Completado";
      }
    }

    function updateGlobalProgress() {
      const count = completedDays.size;
      const pct = Math.round((count / totalDays) * 100);
      document.getElementById('globalProgressText').innerText = `${count} / 365 (${pct}%)`;
    }

    function jumpToDay(val) {
      const d = parseInt(val);
      if (d >= 1 && d <= 365) loadDayContent(d);
    }

    function toggleAudio() {
      if (isPlaying) {
        stopAudio();
      } else {
        playAudio();
      }
    }

    function playAudio() {
      if ('speechSynthesis' in window) {
        window.speechSynthesis.cancel();
        const text = `Estudio Bíblico Día ${currentDay}. Autor Joseph Ibáñez. Reflexión del pastor y oración del día.`;
        speechSyn = new SpeechSynthesisUtterance(text);
        speechSyn.lang = 'es-ES';
        speechSyn.onend = () => stopAudio();
        window.speechSynthesis.speak(speechSyn);
        isPlaying = true;
        document.getElementById('audioPlayIcon').className = "fa-solid fa-pause";
      } else {
        alert("Tu navegador no soporta audio.");
      }
    }

    function stopAudio() {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      isPlaying = false;
      document.getElementById('audioPlayIcon').className = "fa-solid fa-play ml-0.5";
    }

    function seekAudio(val) {}
  </script>
